  /* - Grand Bazaar, rebuilt from Mr Eric's own spice-market photos after the first
     version read as muddy rather than vivid. The problem wasn't the colour choice, it
     was the background: on a light cream page, genuinely vivid spice colours (a real
     pixel-sampled burnt orange, curry gold, olive tea, dried-flower mauve) all measure
     under 3:1 as text and under 4:1 as a white-text tile - not close, real fails - and
     the only way to make them pass on cream is to darken them 50-70%, which is what
     produced the muddy result. Pairing the SAME colours with a dark background instead
     (the actual shadowed-stall look in the reference photos, not a bright page) needs
     only a 10-20% darkening to pass white-text-on-tile, so they stay recognisably
     vivid: burnt orange #DB5B0D -> #C5510B, curry gold #B87A0E -> #93610B, olive
     #8B7429 -> #7D6824, dried-flower mauve #BE6D85 -> #98576A. Sumac #914334 and a
     deep paprika red #6C0D07 needed no adjustment at all.

     This reintroduces the exact same structural conflict Uzbekistan2 hit (--coral/
     --teal/--gold/--coral-dark each used both as light-needing text and dark-needing
     tile fills elsewhere in the codebase) and is resolved the same way: each variable
     set to whichever role dominates its call sites, with an explicit override for
     every call site on the losing side. Same ~30 call sites as Uzbekistan2, same file,
     just Grand-Bazaar colours in place of Uzbekistan2's. */

function setupBazaarMotes(active){
    let field = document.getElementById('bazaar-mote-field');
    if(!active){
      if(field) field.remove();
      const glass = document.getElementById('bazaar-tea-glass');
      if(glass) glass.remove();
      return;
    }
    if(field) return;
    field = document.createElement('div');
    field.id = 'bazaar-mote-field';
    field.setAttribute('aria-hidden', 'true');
    field.style.cssText = 'position:fixed;inset:0;z-index:-1;pointer-events:none;overflow:hidden;';
    document.body.appendChild(field);
    for(let i=0; i<14; i++){
      const m = document.createElement('div');
      m.className = 'bazaar-mote';
      const size = (Math.random()*2.5 + 1.5).toFixed(1);
      const x = (Math.random()*100).toFixed(1);
      const duration = (14 + Math.random()*10).toFixed(1);
      m.style.width = size + 'px';
      m.style.height = size + 'px';
      m.style.left = x + '%';
      m.style.bottom = '-5%';
      m.style.animationDuration = duration + 's';
      m.style.animationDelay = '-' + (Math.random()*duration).toFixed(1) + 's';
      field.appendChild(m);
    }
    const glass = document.createElement('div');
    glass.id = 'bazaar-tea-glass';
    glass.setAttribute('aria-hidden', 'true');
    glass.innerHTML =
      '<div class="bazaar-tea-glass-icon"></div>' +
      '<div class="bazaar-steam" style="left:6px;animation-delay:0s;"></div>' +
      '<div class="bazaar-steam" style="left:11px;animation-delay:-1.6s;"></div>' +
      '<div class="bazaar-steam" style="left:16px;animation-delay:-3.1s;"></div>';
    document.body.appendChild(glass);
  }
