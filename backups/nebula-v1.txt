     Same structural conflict as the other dark-background themes still applies for
     --coral/--teal/--gold/--coral-dark (each used both as light-needing text and dark-
     needing tile fills elsewhere in the codebase) - same ~30 call-site audit, same
     resolution.

     Starfield: scattered white points with an opacity-only twinkle, on both the page
     and the header specifically (the header has its own opaque background, so a
     page-level star layer alone would be invisible behind it - needs its own layer).
     Deliberately opacity-only, no transform/blur/position animation: this codebase has
     a standing lesson on exactly this class of effect - the aurora-layer's original
     animation was removed because "it ran forever, every frame, competing with real
     interactions for the main thread." Opacity fades on a handful of pseudo-elements
     are compositor-handled and cheap; that's the only animation type used here, and the
     star count is kept modest (4 layers, ~12 points each) rather than hundreds of
     independent DOM elements. */

/* ---- JS: setupNebulaStars ---- */
  function setupNebulaStars(active){
    let field = document.getElementById('nebula-star-field');
    if(!active){
      if(field) field.remove();
      document.querySelectorAll('.nebula-header-star').forEach(el => el.remove());
      const ring = document.getElementById('nebula-planet-ring');
      const sphere = document.getElementById('nebula-planet-sphere');
      if(ring) ring.remove();
      if(sphere) sphere.remove();
      document.querySelectorAll('.nebula-shooting-star').forEach(el => el.remove());
      return;
    }
    // Widened size (0.5-3.4px, was 0.8-2.4px) and duration (1.5-7s, was 2-5s) ranges so
    // "some bigger some smaller, some faster some slower" is actually visible rather
    // than a fairly uniform field. Peak opacity is now per-star too (--peak, .45-1) via
    // a CSS custom property read by the keyframe, so brightness varies star-to-star as
    // well, not just size/speed - bigger stars biased toward a higher peak (reads as
    // "closer/brighter"), smaller stars can land dimmer.
    function makeStar(fixedX){
      const s = document.createElement('div');
      s.className = 'nebula-star' + (fixedX !== undefined ? ' nebula-header-star' : '');
      const size = Math.random()*2.9 + 0.5;
      const x = (fixedX !== undefined ? fixedX : Math.random()*100).toFixed(1);
      const y = (Math.random()*100).toFixed(1);
      const duration = (Math.random()*5.5 + 1.5).toFixed(2);
      const peak = (0.45 + (size/3.4)*0.4 + Math.random()*0.15).toFixed(2);
      s.style.width = size.toFixed(1) + 'px';
      s.style.height = size.toFixed(1) + 'px';
      s.style.left = x + '%';
      s.style.top = y + '%';
      s.style.setProperty('--peak', peak);
      s.style.animationDuration = duration + 's';
      s.style.animationDelay = '-' + (Math.random()*duration).toFixed(2) + 's';
      return s;
    }
    if(!field){
      field = document.createElement('div');
      field.id = 'nebula-star-field';
      field.setAttribute('aria-hidden', 'true');
      document.body.appendChild(field);
      for(let i=0; i<59; i++) field.appendChild(makeStar());
    }
    const header = document.querySelector('header');
    if(header && !header.querySelector('.nebula-header-star')){
      for(let i=0; i<10; i++){
        const leftSide = Math.random() < 0.5;
        const x = leftSide ? Math.random()*18 : 82 + Math.random()*18;
        header.appendChild(makeStar(x));
      }
    }
    // Planet + ring: created once, static (no per-frame work), so a plain existence
    // check is enough - no randomisation needed since it's a single fixed element.
    if(!document.getElementById('nebula-planet-sphere')){
      const ring = document.createElement('div');
      ring.id = 'nebula-planet-ring';
      ring.setAttribute('aria-hidden', 'true');
      document.body.appendChild(ring);
      const sphere = document.createElement('div');
      sphere.id = 'nebula-planet-sphere';
      sphere.setAttribute('aria-hidden', 'true');
      document.body.appendChild(sphere);
    }
    // Shooting stars: 3 elements, each on its own long randomised cycle (12-22s) so
    // they fire independently rather than in sync, with a randomised starting position
    // along the top-right so they don't all streak from the same point.
    if(!document.querySelector('.nebula-shooting-star')){
      for(let i=0; i<3; i++){
        const s = document.createElement('div');
        s.className = 'nebula-shooting-star';
        s.setAttribute('aria-hidden', 'true');
        s.style.top = (10 + Math.random()*35).toFixed(1) + '%';
        s.style.right = (5 + Math.random()*30).toFixed(1) + '%';
        const duration = (12 + Math.random()*10).toFixed(1);
        s.style.animationDuration = duration + 's';
        s.style.animationDelay = '-' + (Math.random()*duration).toFixed(1) + 's';
        document.body.appendChild(s);
      }
